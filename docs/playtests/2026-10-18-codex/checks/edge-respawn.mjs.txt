import {pathToFileURL,fileURLToPath} from 'node:url';

const D=path.dirname(fileURLToPath(import.meta.url));

  await p.screenshot({path:D+'/08-lip-fall-repeat'+i+'.png'});
  await p.keyboard.press('KeyP');await p.evaluate(()=>BB.settle(60));const after=await p.evaluate(()=>BB.game());results.push({name:'outer_lip_repeat'+i,mid,after});
 }

 await p.screenshot({path:D+'/09-held-input-after-fullreset.png'});
 console.log(JSON.stringify(results));
}finally{await browser.close();writeFileSync(D+'/edge-respawn.json',JSON.stringify({method:'BB.place fixture establishes stationary inner6.8 or outer7.35 beach coordinates,z-8; physics and damage unchanged.
