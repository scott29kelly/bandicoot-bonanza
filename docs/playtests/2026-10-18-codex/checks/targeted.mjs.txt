import {chromium} from '../snapshot/node_modules/playwright/index.mjs';
import {writeFileSync,mkdirSync} from 'node:fs';
import {pathToFileURL,fileURLToPath} from 'node:url';
import path from 'node:path';
const D=path.dirname(fileURLToPath(import.meta.url)), ROOT=path.resolve(D,'../snapshot');
const results=[],errors=[];
const browser=await chromium.launch({channel:'msedge',headless:true,args:['--mute-audio','--force-color-profile=srgb']});
const page=await browser.newPage({viewport:{width:960,height:600},deviceScaleFactor:1});
page.on('pageerror',e=>errors.push(e.message));
const check=(name,pass,detail)=>{results.push({name,pass:!!pass,detail});console.log(JSON.stringify(results.at(-1)));writeFileSync(D+'/targeted-results.json',JSON.stringify({results,errors},null,2));};
const shot=async name=>page.screenshot({path:D+'/'+name+'.png'});
const boot=async()=>{const u=new URL(pathToFileURL(ROOT+'/index.html'));u.searchParams.set('fixeddt','0.016666666666666666');u.searchParams.set('minfx','1');await page.goto(u.href);await page.waitForFunction(()=>window.BB?.ready,{},{timeout:120000});await page.evaluate(()=>BB.settle(5));};
const reset=async()=>{await page.keyboard.press('KeyP');await page.locator('#restart-game').click();await page.evaluate(()=>BB.settle(5));};
try{
 await boot();
 const jump=await page.evaluate(async()=>{
  const key=(code,type='keydown')=>window.dispatchEvent(new KeyboardEvent(type,{code,bubbles:true}));
  BB.place([0,0,-5]);await BB.settle(3);key('Space');await BB.settle(1);key('Space','keyup');
  const first=[];for(let i=0;i<18;i++){await BB.settle(1);first.push(BB.game().hero[1]);}
  key('Space');await BB.settle(1);key('Space','keyup');const second=[];for(let i=0;i<12;i++){await BB.settle(1);second.push(BB.game().hero[1]);}
  const before=BB.game().hero[1];key('Space');await BB.settle(1);key('Space','keyup');const third=[];for(let i=0;i<4;i++){await BB.settle(1);third.push(BB.game().hero[1]);}
  await BB.settle(90);return{first,second,before,third,final:BB.game()};
 });
 check('jump and double jump gain height',jump.first[0]>0&&jump.second.at(-1)>Math.max(...jump.first),jump);
 check('third jump does not reset upward velocity',jump.third[3]-jump.third[2]<jump.third[1]-jump.third[0],jump.third);
 check('jump returns to original floor without life loss',Math.abs(jump.final.hero[1])<.01&&jump.final.lives===3,jump.final);
 for(const delay of [2,10]){
  const r=await page.evaluate(async delay=>{const key=(code,type='keydown')=>window.dispatchEvent(new KeyboardEvent(type,{code,bubbles:true}));BB.place([6.6,0,-5]);await BB.settle(3);key('KeyD');let left;for(let i=0;i<30;i++){await BB.settle(1);if(BB.game().hero[1]<-.001){left=BB.game();break;}}key('KeyD','keyup');await BB.settle(delay);const before=BB.game();key('Space');await BB.settle(1);key('Space','keyup');await BB.settle(5);return{left,before,after:BB.game()};},delay);
  check(delay===2?'coyote jump works shortly after leaving ledge':'coyote jump expires after grace window',delay===2?r.after.hero[1]>r.before.hero[1]:r.after.hero[1]<r.before.hero[1],r);
  await reset();
 }
 const buffered=await page.evaluate(async()=>{BB.place([0,3,-5]);let queued;for(let i=0;i<50;i++){await BB.settle(1);const y=BB.game().hero[1];if(y>0&&y<.28){queued=BB.game();window.dispatchEvent(new KeyboardEvent('keydown',{code:'Space',bubbles:true}));window.dispatchEvent(new KeyboardEvent('keyup',{code:'Space',bubbles:true}));break;}}await BB.settle(8);return{queued,after:BB.game()};});
 check('jump input shortly before landing buffers into grounded jump',!!buffered.queued&&buffered.after.hero[1]>.5,buffered);
 await reset();
 const input=await page.evaluate(async()=>{const key=(code,type='keydown')=>window.dispatchEvent(new KeyboardEvent(type,{code,bubbles:true}));key('KeyW');await BB.settle(10);key('KeyP');key('KeyP','keyup');const paused=BB.game();key('Space');key('KeyK');await BB.settle(20);return{paused,after:BB.game()};});
 check('pause freezes movement and ignores jump/spin input',JSON.stringify(input.paused)===JSON.stringify(input.after),input);
 await page.locator('#resume-game').click();await page.evaluate(()=>BB.settle(50));const resumed=await page.evaluate(()=>BB.game());
 check('resume discards held movement and queued actions',Math.abs(resumed.hero[1])<.01&&Math.abs(resumed.hero[2]-input.paused.hero[2])<1,resumed);
 await reset();
 await page.evaluate(()=>{BB.place([0,0,-50]);BB.setPose({p:[6,3,-46],lookAt:[0,0.7,-52],fov:55});return BB.settle(20)});await shot('01-yard-before-spin');
 await page.keyboard.press('KeyK');await page.evaluate(()=>BB.settle(10));await shot('02-yard-armed');
 const armed=await page.evaluate(()=>BB.game());
 await page.keyboard.press('KeyP');const frozen=await page.evaluate(()=>BB.game());await page.evaluate(()=>BB.settle(150));const frozenAfter=await page.evaluate(()=>BB.game());
 check('armed TNT fuse and lives freeze while paused',armed.tnt[0].armed&&frozen.tnt[0].fuse===frozenAfter.tnt[0].fuse&&frozen.lives===frozenAfter.lives,{armed,frozen,frozenAfter});
 await page.locator('#resume-game').click();await page.evaluate(()=>BB.settle(150));const explosion=await page.evaluate(()=>BB.game());
 check('TNT at blast range costs exactly one life',explosion.lives===2&&explosion.deaths===1&&explosion.tnt[0].exploded,explosion);
 await page.evaluate(()=>{BB.place([0,0,-49]);return BB.settle(30)});await shot('03-yard-after-tnt');await page.evaluate(()=>BB.clearPose());
 check('spent TNT and destroyed crates stay gone after respawn',explosion.tnt[0].exploded&&explosion.crates===7,explosion);
 await reset();const clean=await page.evaluate(()=>BB.game());
 check('Restart run restores TNT crates fruit lives and checkpoint',clean.lives===3&&clean.crates===0&&clean.cratesLeft===9&&clean.fruit===0&&clean.checkpoint===0&&!clean.tnt[0].exploded&&!clean.tnt[0].armed,clean);
 for(const width of [960,640,320]){
  await page.setViewportSize({width,height:width===320?568:360});await page.evaluate(()=>BB.settle(5));await page.keyboard.press('KeyP');
  const box=await page.evaluate(()=>{const r=document.querySelector('.pause-card').getBoundingClientRect();return{x:r.x,y:r.y,w:r.width,h:r.height,viewport:[innerWidth,innerHeight],scrollH:document.querySelector('.pause-card').scrollHeight,clientH:document.querySelector('.pause-card').clientHeight};});await shot('04-pause-'+width);
  check('pause card fits viewport '+width,box.x>=0&&box.y>=0&&box.x+box.w<=width+1&&box.y+box.h<=box.viewport[1]+1,box);
  await page.locator('#resume-game').click();
 }
 await page.setViewportSize({width:960,height:600});
 const edge=await page.evaluate(async()=>{BB.place([7.35,0,-8]);await BB.settle(4);window.dispatchEvent(new KeyboardEvent('keydown',{code:'KeyP',bubbles:true}));return BB.game();});
 await page.evaluate(()=>document.querySelector('#pause-panel').classList.add('hidden'));await shot('05-sand-edge-fall');
 check('diagnostic: nominal collision ends before decorative sand lip',edge.hero[1]<0,edge);
 check('no runtime errors during targeted checks',errors.length===0&&!await page.evaluate(()=>BB.error),errors);
}finally{await browser.close();writeFileSync(D+'/targeted-results.json',JSON.stringify({method:'Playwright Edge headless local snapshot. Fixed1/60 simulation/minfx. BB.place used for isolated setup; gameplay input and timers remain active. Screenshot after edge pauses simulation and hides pause overlay only. No source edits.',results,errors},null,2));}
