import {chromium} from 'file:///C:/Users/scott/Documents/repos/games/bandicoot-bonanza/main/node_modules/playwright/index.mjs';
import {mkdirSync,writeFileSync,existsSync,readFileSync,unlinkSync} from 'node:fs';
import {pathToFileURL} from 'node:url';
const E='C:/Users/scott/.codex/visualizations/2026/09/29/01a0ef07-70d9-7850-baa4-9a3ce54831a0/bandicoot-comprehensive-20261018';
const D=E+'/route';let log=[],errors=[];
const browser=await chromium.launch({channel:'msedge',headless:true,args:['--mute-audio','--force-color-profile=srgb','--hide-scrollbars']});
const page=await browser.newPage({viewport:{width:1280,height:720},deviceScaleFactor:1});
page.on('pageerror',e=>errors.push(e.message));page.on('console',m=>{if(m.type()==='error')errors.push(m.text())});
const url=new URL(pathToFileURL(E+'/snapshot/index.html'));url.searchParams.set('fixeddt','0.016666666666666666');
await page.goto(url.href,{waitUntil:'domcontentloaded'});await page.waitForFunction(()=>window.BB?.ready,{},{timeout:120000});
await page.evaluate(()=>BB.settle(10));
const state=async()=>page.evaluate(()=>({game:BB.game(),frame:BB.stats.frame,error:BB.error,status:document.getElementById('game-status').textContent,paused:!document.getElementById('pause-panel').classList.contains('hidden')}));
console.log('READY '+JSON.stringify(await state()));
while(true){
 if(!existsSync(D+'/command.json')){await new Promise(r=>setTimeout(r,200));continue;}
 let c=JSON.parse(readFileSync(D+'/command.json','utf8'));unlinkSync(D+'/command.json');
 if(c.quit)break;
 const before=await state();
 try{
  for(const a of c.actions||[]){
   if(a.down)await page.keyboard.down(a.down);
   if(a.up)await page.keyboard.up(a.up);
   if(a.press)await page.keyboard.press(a.press);
   if(a.frames)await page.evaluate(n=>BB.settle(n),a.frames);
   if(a.click)await page.locator(a.click).click();
   if(a.shot)await page.screenshot({path:D+'/'+a.shot+'.png'});
  }
  const after=await state();log.push({name:c.name,before,actions:c.actions,after});
  writeFileSync(D+'/trace.json',JSON.stringify({url:url.href,method:'Native Playwright keyboard input, BB.settle fixed 1/60 dt and read-only BB.game; no place/review/setPose/state mutation.',log,errors},null,2));
  console.log(c.name+' '+JSON.stringify(after));
  writeFileSync(D+'/result.json',JSON.stringify({name:c.name,...after,errors},null,2));
 }catch(e){console.log('ERROR '+e.stack);writeFileSync(D+'/result.json',JSON.stringify({name:c.name,error:e.stack}));}
}
await browser.close();
